let marks = Number(prompt("Enter student marks:"));
let result;

if (marks >= 80) {
    result = "A+";
}
else if (marks >= 70) {
    result = "A";
}
else if (marks >= 60) {
    result = "B";
}
else {
    result = "F";
}

console.log("Marks:", marks);
console.log("Calculated Grade:", result);
